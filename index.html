Papa.parse("palpites.csv", { download: true, header: true, skipEmptyLines: true, complete: function(res) { let html = ""; res.data.forEach(row => { if(!row.Confronto) return; const times = row.Confronto.split(" vs "); // Agora ele lê 'Entrada' que o Python gerou const palpiteMestre = row.Entrada || "Analisando..."; const confMestre = row.Confianca || "90%"; html += `
${row.Hora || '00:00'} CONFIRMADO
${times[0]}
|
${times[1]}
ENTRADA RECOMENDADA ${palpiteMestre}
${confMestre}
`; }); document.getElementById("lista-jogos").innerHTML = html; } });
